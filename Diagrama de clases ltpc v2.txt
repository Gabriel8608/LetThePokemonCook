================================================================================
FLUJOS PRINCIPALES
================================================================================

FLUJO "RECETA ALEATORIA":
  MainActivity → MainViewModel.fetchMealWithPokemon()
    → MealApiService.getRandomMeal()  [HTTP themealdb.com]
    → PokeApiService.getPokemon(random)  [HTTP pokeapi.co]
    → RecipeUiModel
    → LiveData<RecipeUiModel?> → RecipeScreen

FLUJO "POKEDEX":
  MainActivity → PokemonListScreen
    → PokemonListViewModel.uiState (StateFlow)
    → PokemonRepository.obtenerPokemones()
    → PokeApiService.getPokemon(1..20)  [HTTP pokeapi.co]
    → List<PokemonResumen>
    → PokemonListUiState

FLUJO "FAVORITOS":
  PokemonCard.onFavoritoClick
    → PokemonListViewModel.alternarFavorito(pokemon)
    → FavoritosRepository.agregar/quitar(id)
    → FavoritoDao (Room)  ⇄  FavoritoEntity  ⇄  SQLite "pokedex.db"
    → Flow<List<Int>> → favoritosIds → UI actualiza corazones

FLUJO "DETALLE":
  PokemonListScreen → onPokemonClick(pokemon)
    → MainViewModel.verDetallePokemon(nombre)
    → AppScreen.DETALLE
    → PokemonDetailScreen → PokemonDetailViewModel
    → PokemonRepository.obtenerDetalle(nombre)
    → PokeApiService.getPokemon(id)
    → PokemonDetalle
    → PokemonDetailUiState
